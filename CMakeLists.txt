cmake_minimum_required(VERSION 3.14)
project(libana LANGUAGES CXX)

# Force modern C++20 optimization and paradigms
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# -------------------------------------------------------------------
# Third-Party Libraries Installation (Automatic Fetching)
# -------------------------------------------------------------------
include(FetchContent)

# CRITICAL: Disable Eigen's internal test/demo suite configuration
set(EIGEN_BUILD_TESTING OFF CACHE BOOL "Disable Eigen testing" FORCE)
set(EIGEN_BUILD_DOC OFF CACHE BOOL "Disable Eigen documentation" FORCE)

# Fetching stable Eigen library for sparse matrix operations
FetchContent_Declare(
  Eigen
  GIT_REPOSITORY https://gitlab.com/libeigen/eigen.git
  GIT_TAG        3.4.0
)
FetchContent_MakeAvailable(Eigen)

# -------------------------------------------------------------------
# Target Resolution & Discovery
# -------------------------------------------------------------------
file(GLOB_RECURSE LIB_SOURCES 
    "src/core/*.cpp"
    "src/solver/*.cpp"
)

# Core Playground and Simulation target
add_executable(ana_sim src/main.cpp ${LIB_SOURCES})

# Include Paths Setup
target_include_directories(ana_sim PRIVATE src)

# Link against the correct internal header target generated by add_subdirectory
target_link_libraries(ana_sim PRIVATE eigen)

# -------------------------------------------------------------------
# Include Nested Directories
# -------------------------------------------------------------------
add_subdirectory(third_party)
